# Replaces kendryte-standalone-sdk/lib/nncase/v1/CMakeLists.txt after
# build.sh swaps in nncase's v1.9.0 K210 runtime. The stock file uses
# find_package(nncaseruntime), but the 1.9.0 package's cmake config declares
# an imported target named `kendryte`, which collides with the SDK's own
# `kendryte` target -- so link the .a files directly instead. The nncase
# 1.9.0 headers also include <gsl/gsl-lite.hpp>, which the SDK's header glob
# no longer puts on this target's include path, so add the third_party roots.
set(NNCASE_V1_LIBDIR ${CMAKE_CURRENT_LIST_DIR}/lib)
set(NNCASE_V1_TP ${CMAKE_CURRENT_LIST_DIR}/../../../third_party)

add_library(nncase-v1 STATIC nncase_v1.cpp)
target_include_directories(nncase-v1 PRIVATE
  ${CMAKE_CURRENT_LIST_DIR}/include
  ${NNCASE_V1_TP}/gsl-lite/include
  ${NNCASE_V1_TP}/mpark-variant/include
  ${NNCASE_V1_TP}/nlohmann_json/include
  ${NNCASE_V1_TP}/xtl/include)
target_link_libraries(nncase-v1 PRIVATE
  -Wl,-start-group
  ${NNCASE_V1_LIBDIR}/libnncase.runtime.a
  ${NNCASE_V1_LIBDIR}/libnncase.rt_modules.k210.a
  ${NNCASE_V1_LIBDIR}/libnncase.functional.a
  -Wl,-end-group)
